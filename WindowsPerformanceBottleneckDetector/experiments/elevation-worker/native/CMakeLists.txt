cmake_minimum_required(VERSION 3.24)
project(WpbdNativeRefusalPrototype LANGUAGES CXX)
if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "Reviewed Windows x64 MSVC/SDK toolchain required. No tool installation or privilege fallback.")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_executable(WpbdNativeWorker Worker.cpp)
target_link_libraries(WpbdNativeWorker PRIVATE advapi32)
target_compile_options(WpbdNativeWorker PRIVATE /W4 /WX /sdl /guard:cf /utf-8)
target_link_options(WpbdNativeWorker PRIVATE /DYNAMICBASE /HIGHENTROPYVA /NXCOMPAT /guard:cf /CETCOMPAT /DEPENDENTLOADFLAG:0x800
  /MANIFEST:EMBED "/MANIFESTINPUT:${CMAKE_CURRENT_SOURCE_DIR}/../asInvoker.manifest")
add_executable(WpbdNativeRequestTests RequestTests.cpp)
target_compile_options(WpbdNativeRequestTests PRIVATE /W4 /WX /sdl /utf-8)
enable_testing()
add_test(NAME NativeRequestGrammar COMMAND WpbdNativeRequestTests)
